match /qrajn/branding/{uid}/logo {
  allow read: if true;
  allow write: if request.auth != null
               && request.auth.uid == uid
               && request.resource != null
               && request.resource.size < 1024 * 1024
               && request.resource.contentType.matches('image/(png|jpeg|webp)');
}
