function qrajnSignedIn() {
  return request.auth != null;
}

function qrajnOwner(uid) {
  return qrajnSignedIn() && request.auth.uid == uid;
}

function qrajnImage(maxBytes) {
  return request.resource != null
    && request.resource.size <= maxBytes
    && request.resource.contentType != null
    && request.resource.contentType.matches('image/(png|jpeg|webp)');
}

function qrajnPdf(maxBytes) {
  return request.resource != null
    && request.resource.size <= maxBytes
    && request.resource.contentType == 'application/pdf';
}

  allow delete: if qrajnOwner(uid);
